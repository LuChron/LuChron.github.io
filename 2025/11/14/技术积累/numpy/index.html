# NumPy 是什么

NumPy (Numerical Python) 是 Python 做科学计算的基础库。
核心是一个高性能的 多维数组对象 ndarray，比 Python 自带的 list 更快、更省内存。

# 导入与数组创建

import numpy as np

# 创建数组

a = np.array([1, 2, 3])              # 一维数组
b = np.array([[1, 2, 3], [4, 5, 6]]) # 二维数组

# 创建全 0、全 1、空数组

np.zeros((2,3))      # [[0,0,0],[0,0,0]]
np.ones((2,3))       # [[1,1,1],[1,1,1]]
np.empty((2,3))      # 未初始化的值（随机内存内容）

# 数值范围

np.arange(0, 10, 2)  # [0, 2, 4, 6, 8] 根据步长生成等差数列
np.linspace(0, 1, 5) # [0., 0.25, 0.5, 0.75, 1.] 根据元素个数生成等差数列

# 随机数

np.random.rand(2,3)       # 2*3矩阵[0,1) 之间均匀分布
np.random.randn(2,3)      # 2*3矩阵标准正态分布
np.random.randint(0,10,5) # 0~9 的随机整数 生成5个整数，范围是(0,9]

# 数组基本属性

a = np.array([[1, 2, 3], [4, 5, 6]])

a.ndim   # 维度（2）
a.shape  # 形状 (2,3)
a.size   # 元素总数 (6)
a.dtype  # 数据类型（int64, float32等）
a.itemsize # 每个元素字节数

# 索引与切片

# 普通索引

a[0, 1]   # 第1行第2列的元素
a[:, 1]   # 取第2列 [2, 5]
a[1, :]   # 取第2行 [4, 5, 6]

# 布尔索引

a[a > 3]     # 返回所有 >3 的元素

# 花式索引（Fancy Indexing）

a[[0,1], [1,2]]  # 提取 (0,1) 和 (1,2) 的元素

# 基本运算

# 数组与标量运算（广播机制）

a = np.array([1,2,3])
a + 10      # [11,12,13]
a * 2       # [2,4,6]

# 数组与数组运算

b = np.array([4,5,6])
a + b       # [5,7,9]
a * b       # [4,10,18]（逐元素相乘）

# 常见函数

np.sqrt(a) # 开根
np.exp(a)  # 几次方
np.sin(a)  # sin
np.log(a)  # ln

# 聚合函数

a.sum()
a.mean()
a.max()
a.min()
a.std()  # 标准差
a.var()  # 方差
a.argmin()  # 最小值索引
a.argmax()  # 最大值索引

# 矩阵运算

A = np.array([[1,2],[3,4]])
B = np.array([[5,6],[7,8]])

A @ B          # 矩阵乘法（推荐）
np.dot(A, B)   # 同上
A.T            # 转置
np.linalg.inv(A)  # 逆矩阵
np.linalg.det(A)  # 行列式
np.linalg.eig(A)  # 特征值和特征向量

# 数组变形与拼接

# 变形

a.reshape(3,2)
a.ravel()     # 拉平成一维       这个是可能会影响原来的数组
a.flatten()   # 同上，返回拷贝    这个不会影响原来的数组，就是相当于复制一份新的

# 拼接

np.hstack((a,b))   # 水平拼接  相当于两个一左一右放置
np.vstack((a,b))   # 垂直拼接  相当于两个一上一下放置
np.concatenate((a,b), axis=0)  # 0是上下拼 1是左右拼

# 拆分

np.hsplit(a, 2) # 水平切成左右两块
np.vsplit(a, 2) # 竖直切成上下两块  
# 如果是奇数的话切割会报错

# 数据类型转换

a.astype(float)   # 转换为浮点型
a.astype(np.int32)

# 文件操作

np.save('data.npy', a) # 把数组保存为.npy的二进制文件
b = np.load('data.npy') # 从二进制文件中还原出原始数组

np.savetxt('data.txt', a, fmt='%d') # 保存为文本文件格式，可读但是文件会更大，后面这个是保存控制格式
b = np.loadtxt('data.txt') # 从文本中恢复数组

# 实用技巧

np.unique(a)                # 去重
np.sort(a)                  # 排序
np.argsort(a)               # 排序索引
np.clip(a, 0, 10)           # 限制范围到 [0,10]
np.where(a > 5, 1, 0)       # 条件替换
np.all(a > 0)               # 全部为真？
np.any(a > 0)               # 任意为真？

# 性能对比（与 Python list）

import time

a = np.arange(10**6)
b = np.arange(10**6)

t0 = time.time()
a + b
print(time.time() - t0)  # NumPy 非常快！

# list版本：
L1 = list(range(10**6))
L2 = list(range(10**6))
t0 = time.time()
[L1[i] + L2[i] for i in range(10**6)]
print(time.time() - t0)  # 慢得多

# 总结表

分类 常用函数

运算 + - * /, sum, mean, max, min
线代 dot, inv, det, eig, T
索引 [ ], :, bool索引, fancy索引
变形 reshape, ravel, hstack, vstack
IO save, load, savetxt, loadtxt
